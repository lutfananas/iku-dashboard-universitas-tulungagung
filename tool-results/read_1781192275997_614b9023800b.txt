    66→// ============ MAIN COMPONENT ============
    67→export default function IKUDashboard() {
    68→  const [navSelection, setNavSelection] = useState<NavSelection>({ mode: "global", id: "global" });
    69→  const [tahun, setTahun] = useState(2025);
    70→  const [allData, setAllData] = useState<IkuRecord[]>([]);
    71→  const [loading, setLoading] = useState(true);
    72→  const [expandedFakultas, setExpandedFakultas] = useState<Set<string>>(new Set());
    73→  const [saving, setSaving] = useState<string | null>(null);
    74→  const [saveSuccess, setSaveSuccess] = useState<string | null>(null);
    75→
    76→  // Form state for current prodi+tahun
    77→  const [formData, setFormData] = useState<Record<string, Record<string, number | string>>>({});
    78→
    79→  // Fetch data
    80→  const fetchData = useCallback(async () => {
    81→    try {
    82→      const res = await fetch(`/api/iku?tahun=${tahun}`);
    83→      const data = await res.json();
    84→      setAllData(data);
    85→    } catch (e) {
    86→      console.error("Fetch error:", e);
    87→    } finally {
    88→      setLoading(false);
    89→    }
    90→  }, [tahun]);
    91→
    92→  useEffect(() => { fetchData(); }, [fetchData]);
    93→
    94→  // Get data for a specific prodi
    95→  const getProdiData = useCallback((prodiId: string): IkuRecord | undefined => {
    96→    return allData.find((d) => d.prodi === prodiId);
    97→  }, [allData]);
    98→
    99→  // Initialize form data when prodi/tahun changes
   100→  useEffect(() => {
   101→    if (navSelection.mode !== "prodi") return;
   102→    const prodiData = getProdiData(navSelection.id);
   103→    const newFormData: Record<string, Record<string, number | string>> = {};
   104→
   105→    IKU_LIST.forEach((iku) => {
   106→      const existingData = prodiData?.[iku.id as keyof IkuRecord] as Record<string, unknown> | null;
   107→      const fields = IKU_FIELDS[iku.id];
   108→      const ikuFormData: Record<string, number | string> = {};
   109→      fields.forEach((field) => {
   110→        if (existingData && existingData[field.key] !== undefined && existingData[field.key] !== null) {
   111→          ikuFormData[field.key] = existingData[field.key] as number | string;
   112→        } else {
   113→          ikuFormData[field.key] = field.defaultValue ?? (field.type === "number" ? 0 : "");
   114→        }
   115→      });
   116→      newFormData[iku.id] = ikuFormData;
   117→    });
   118→
   119→    setFormData(newFormData);
   120→  }, [navSelection, tahun, allData, getProdiData]);
   121→
   122→  // Save handler
   123→  const handleSave = async (ikuId: string) => {
   124→    if (navSelection.mode !== "prodi") return;
   125→    const prodi = getProdiById(navSelection.id);
   126→    if (!prodi) return;
   127→
   128→    setSaving(ikuId);
   129→    try {
   130→      const res = await fetch("/api/iku", {
   131→        method: "POST",
   132→        headers: { "Content-Type": "application/json" },
   133→        body: JSON.stringify({
   134→          prodi: prodi.id,
   135→          fakultas: prodi.fakultasId,
   136→          tahun,
   137→          ikuId,
   138→          data: formData[ikuId],
   139→        }),
   140→      });
   141→      if (res.ok) {
   142→        setSaveSuccess(ikuId);
   143→        setTimeout(() => setSaveSuccess(null), 2000);
   144→        await fetchData();
   145→        toast.success("Data berhasil disimpan", { description: `${IKU_LIST.find((i) => i.id === ikuId)?.label} - ${prodi.nama} (${tahun})` });
   146→      }
   147→    } catch (e) {
   148→      console.error("Save error:", e);
   149→    } finally {
   150→      setSaving(null);
   151→    }
   152→  };
   153→
   154→  // Form input change
   155→  const handleFormChange = (ikuId: string, key: string, value: number | string) => {
   156→    setFormData((prev) => ({
   157→      ...prev,
   158→      [ikuId]: { ...prev[ikuId], [key]: value },
   159→    }));
   160→  };
   161→
   162→  // Toggle fakultas expansion
   163→  const toggleFakultas = (fakultasId: string) => {
   164→    setExpandedFakultas((prev) => {
   165→      const next = new Set(prev);
   166→      if (next.has(fakultasId)) next.delete(fakultasId);
   167→      else next.add(fakultasId);
   168→      return next;
   169→    });
   170→  };
   171→
   172→  // ============ CALCULATION HELPERS ============
   173→  const calcProdiValue = (ikuId: string, prodiId: string): number => {
   174→    const prodiData = getProdiData(prodiId);
   175→    if (!prodiData) return 0;
   176→    const data = prodiData[ikuId as keyof IkuRecord];
   177→    if (!data || !hasData(data as Record<string, unknown>)) return 0;
   178→
   179→    const prodi = getProdiById(prodiId);
   180→    switch (ikuId) {
   181→      case "iku1": return calcAEE_AchievementRate(data as Iku1Data, prodi?.jenjang || "S1");
   182→      case "iku2": return calcIKU2(data as Iku2Data);
   183→      case "iku3": return calcIKU3(data as Iku3Data);
   184→      case "iku5": return calcIKU5(data as Iku5Data);
   185→      case "iku7": return calcIKU7(data as Iku7Data);
   186→      case "iku9": return calcIKU9(data as Iku9Data);
   187→      case "iku12": return calcIKU12_Skor(data as Iku12Data);
   188→      default: return 0;
   189→    }
   190→  };
   191→
   192→  const calcAggregatedValue = (ikuId: string, prodiIds: string[]): number => {
   193→    const dataList: Record<string, unknown>[] = [];
   194→    const prodiJenjangs: string[] = [];
   195→
   196→    prodiIds.forEach((pid) => {
   197→      const prodiData = getProdiData(pid);
   198→      if (prodiData) {
   199→        const data = prodiData[ikuId as keyof IkuRecord];
   200→        if (data && hasData(data as Record<string, unknown>)) {
   201→          dataList.push(data as Record<string, unknown>);
   202→          const prodi = getProdiById(pid);
   203→          if (prodi) prodiJenjangs.push(prodi.jenjang);
   204→        }
   205→      }
   206→    });
   207→
   208→    if (dataList.length === 0) return 0;
   209→
   210→    switch (ikuId) {
   211→      case "iku1": {
   212→        const validData = dataList.map((d, i) => ({ data: d as Iku1Data, jenjang: prodiJenjangs[i] || "S1" }));
   213→        return calcAEE_PT(validData);
   214→      }
   215→      case "iku2": return calcIKU2_Aggregate(dataList as Iku2Data[]);
   216→      case "iku3": return calcIKU3_Aggregate(dataList as Iku3Data[]);
   217→      case "iku5": return calcIKU5_Aggregate(dataList as Iku5Data[]);
   218→      case "iku7": return calcIKU7_Aggregate(dataList as Iku7Data[]);
   219→      case "iku9": return calcIKU9_Aggregate(dataList as Iku9Data[]);
   220→      case "iku12": return calcIKU12_Aggregate(dataList as Iku12Data[]);
   221→      default: return 0;
   222→    }
   223→  };
   224→
   225→  const getProdiHasData = (prodiId: string): boolean => {
   226→    const d = getProdiData(prodiId);
   227→    if (!d) return false;
   228→    return IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>));
   229→  };
   230→
   231→  const getFakultasHasData = (fakultasId: string): boolean => {
   232→    return getProdiByFakultas(fakultasId).some((p) => getProdiHasData(p.id));
   233→  };
   234→
   235→  // ============ RENDER: SIDEBAR ============
   236→  const renderSidebar = () => (
   237→    <aside className="w-64 lg:w-72 shrink-0 bg-white border-r border-slate-200 overflow-y-auto custom-scrollbar">
   238→      {/* University Header */}
   239→      <div className="p-4 bg-navy text-white">
   240→        <div className="flex items-center gap-2 mb-1">
   241→          <Building2 className="w-5 h-5 text-gold" />
   242→          <span className="font-bold text-sm">Universitas Tulungagung</span>
   243→        </div>
   244→        <p className="text-xs text-white/60">Dashboard IKU Perguruan Tinggi</p>
   245→      </div>
   246→
   247→      {/* Year Selector */}
   248→      <div className="p-3 border-b border-slate-100">
   249→        <Label className="text-xs font-medium text-slate-500 mb-1.5 block">Tahun Akademik</Label>
   250→        <Select value={tahun.toString()} onValueChange={(v) => setTahun(parseInt(v))}>
   251→          <SelectTrigger className="h-9 text-sm border-slate-200">
   252→            <Calendar className="w-3.5 h-3.5 mr-1.5 text-slate-400" />
   253→            <SelectValue />
   254→          </SelectTrigger>
   255→          <SelectContent>
   256→            {[2025, 2026, 2027, 2028, 2029, 2030].map((y) => (
   257→              <SelectItem key={y} value={y.toString()}>{y}</SelectItem>
   258→            ))}
   259→          </SelectContent>
   260→        </Select>
   261→      </div>
   262→
   263→      {/* Navigation Tree */}
   264→      <nav className="p-2">
   265→        {/* Global */}
   266→        <button
   267→          onClick={() => setNavSelection({ mode: "global", id: "global" })}
   268→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-all ${
   269→            navSelection.mode === "global"
   270→              ? "bg-navy text-white shadow-md"
   271→              : "text-slate-600 hover:bg-slate-50"
   272→          }`}
   273→        >
   274→          <BarChart3 className="w-4 h-4 shrink-0" />
   275→          <span>Global Universitas</span>
   276→          {allData.length > 0 && (
   277→            <Badge className="ml-auto text-[10px] bg-teal text-white px-1.5 py-0">
   278→              {allData.filter((d) => IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>))).length} prodi
   279→            </Badge>
   280→          )}
   281→        </button>
   282→
   283→        <Separator className="my-2" />
   284→
   285→        {/* Fakultas List */}
   286→        {FAKULTAS_LIST.map((fakultas) => {
   287→          const isExpanded = expandedFakultas.has(fakultas.id);
   288→          const isSelected = navSelection.mode === "fakultas" && navSelection.id === fakultas.id;
   289→          const hasD = getFakultasHasData(fakultas.id);
   290→
   291→          return (
   292→            <div key={fakultas.id} className="mb-0.5">
   293→              <button
   294→                onClick={() => {
   295→                  toggleFakultas(fakultas.id);
   296→                  setNavSelection({ mode: "fakultas", id: fakultas.id });
   297→                }}
   298→                className={`w-full flex items-center gap-2 px-3 py-2 rounded-lg text-sm transition-all ${
   299→                  isSelected ? "bg-navy/10 text-navy font-semibold" : "text-slate-600 hover:bg-slate-50"
   300→                }`}
   301→              >
   302→                {isExpanded ? <ChevronDown className="w-3.5 h-3.5 shrink-0" /> : <ChevronRight className="w-3.5 h-3.5 shrink-0" />}
   303→                <span className="truncate flex-1 text-left">{fakultas.nama.replace("Fakultas ", "F. ")}</span>
   304→                {hasD && <div className="w-2 h-2 rounded-full bg-teal shrink-0" />}
   305→              </button>
   306→
   307→              {isExpanded && (
   308→                <div className="ml-5 mt-0.5 space-y-0.5">
   309→                  {fakultas.prodiList.map((prodi) => {
   310→                    const isProdiSelected = navSelection.mode === "prodi" && navSelection.id === prodi.id;
   311→                    const prodiHasD = getProdiHasData(prodi.id);
   312→                    return (
   313→                      <button
   314→                        key={prodi.id}
   315→                        onClick={() => setNavSelection({ mode: "prodi", id: prodi.id })}
   316→                        className={`w-full flex items-center gap-2 px-3 py-1.5 rounded-md text-xs transition-all ${
   317→                          isProdiSelected
   318→                            ? "bg-navy text-white font-medium"
   319→                            : "text-slate-500 hover:bg-slate-50 hover:text-slate-700"
   320→                        }`}
   321→                      >
   322→                        <GraduationCap className="w-3.5 h-3.5 shrink-0" />
   323→                        <span className="truncate flex-1 text-left">{prodi.nama}</span>
   324→                        {prodiHasD && <div className="w-1.5 h-1.5 rounded-full bg-teal shrink-0" />}
   325→                      </button>
   326→                    );
   327→                  })}
   328→                </div>
   329→              )}
   330→            </div>
   331→          );
   332→        })}
   333→      </nav>
   334→    </aside>
   335→  );
   336→
   337→  // ============ RENDER: IKU FORM FOR PRODI ============
   338→  const renderProdiForm = () => {
   339→    const prodi = getProdiById(navSelection.id);
   340→    if (!prodi) return null;
   341→    const fakultas = getFakultasById(prodi.fakultasId);
   342→
   343→    return (
   344→      <div className="space-y-6">
   345→        {/* Header */}
   346→        <Card className="border-l-4 border-l-navy">
   347→          <CardContent className="p-5">
   348→            <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">
   349→              <div>
   350→                <h2 className="text-lg font-bold text-navy">{prodi.nama}</h2>
   351→                <p className="text-sm text-slate-500">{fakultas?.nama} • Tahun Akademik {tahun}</p>
   352→              </div>
   353→              <Badge variant="outline" className="w-fit text-xs">{prodi.jenjang}</Badge>
   354→            </div>
   355→          </CardContent>
   356→        </Card>
   357→
   358→        {/* IKU Tabs */}
   359→        <Tabs defaultValue="iku1" className="space-y-4">
   360→          <TabsList className="iku-tabs w-full flex h-auto p-1 bg-slate-100/80 rounded-lg gap-0.5 overflow-x-auto">
   361→            {IKU_LIST.map((iku) => (
   362→              <TabsTrigger key={iku.id} value={iku.id} className="text-xs font-semibold px-3 py-2 rounded-md whitespace-nowrap data-[state=active]:shadow-md transition-all">
   363→                {iku.label}
   364→              </TabsTrigger>
   365→            ))}
   366→          </TabsList>
   367→
   368→          {IKU_LIST.map((iku) => {
   369→            const fields = IKU_FIELDS[iku.id];
   370→            const currentData = formData[iku.id] || {};
   371→            const isSaving = saving === iku.id;
   372→            const isSuccess = saveSuccess === iku.id;
   373→
   374→            // Calculate preview
   375→            let previewValue = 0;
   376→            let previewLabel = "";
   377→            try {
   378→              switch (iku.id) {
   379→                case "iku1": {
   380→                  const d = currentData as unknown as Iku1Data;
   381→                  previewValue = calcAEE_Realisation(d);
   382→                  previewLabel = `AEE Realisasi: ${previewValue.toFixed(2)}% | Tingkat Pencapaian: ${calcAEE_AchievementRate(d, prodi.jenjang).toFixed(2)}%`;
   383→                  previewValue = calcAEE_AchievementRate(d, prodi.jenjang);
   384→                  break;
   385→                }
   386→                case "iku2": {
   387→                  previewValue = calcIKU2(currentData as unknown as Iku2Data);
   388→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   389→                  break;
   390→                }
   391→                case "iku3": {
   392→                  previewValue = calcIKU3(currentData as unknown as Iku3Data);
   393→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   394→                  break;
   395→                }
   396→                case "iku5": {
   397→                  previewValue = calcIKU5(currentData as unknown as Iku5Data);
   398→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   399→                  break;
   400→                }
   401→                case "iku7": {
   402→                  previewValue = calcIKU7(currentData as unknown as Iku7Data);
   403→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   404→                  break;
   405→                }
   406→                case "iku9": {
   407→                  previewValue = calcIKU9(currentData as unknown as Iku9Data);
   408→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   409→                  break;
   410→                }
   411→                case "iku12": {
   412→                  previewValue = calcIKU12_Skor(currentData as unknown as Iku12Data);
   413→                  previewLabel = `Skor: ${previewValue.toFixed(2)}`;
   414→                  break;
   415→                }
   416→              }
   417→            } catch { /* empty */ }
   418→
   419→            return (
   420→              <TabsContent key={iku.id} value={iku.id} className="mt-4">
   421→                <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   422→                  {/* Input Form */}
   423→                  <Card className="lg:col-span-2 border border-slate-200">
   424→                    <CardHeader className="pb-3">
   425→                      <CardTitle className="text-base font-semibold text-navy">{iku.title}</CardTitle>
   426→                      <CardDescription className="text-xs">Masukkan data untuk {prodi.nama} tahun {tahun}</CardDescription>
   427→                    </CardHeader>
   428→                    <CardContent className="space-y-4">
   429→                      {iku.id === "iku1" && (
   430→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   431→                          <div className="flex items-start gap-2">
   432→                            <Info className="w-4 h-4 text-navy mt-0.5 shrink-0" />
   433→                            <div className="text-xs text-slate-600">
   434→                              <p className="font-semibold text-navy mb-1">Rumus IKU 1 (AEE PT):</p>
   435→                              <p>AEE = (Lulus Tepat Waktu / Total Masuk) × 100%</p>
   436→                              <p>Tingkat Pencapaian = (AEE Realisasi / AEE Ideal) × 100%</p>
   437→                              <p>AEE Ideal {prodi.jenjang} = {(() => { const v = { D3: 33, D4: 25, S1: 25, S2: 50, S3: 33 } as Record<string, number>; return v[prodi.jenjang] || 25; })()}%</p>
   438→                            </div>
   439→                          </div>
   440→                        </div>
   441→                      )}
   442→                      {iku.id === "iku2" && (
   443→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   444→                          <div className="flex items-start gap-2">
   445→                            <Info className="w-4 h-4 text-navy mt-0.5 shrink-0" />
   446→                            <div className="text-xs text-slate-600">
   447→                              <p className="font-semibold text-navy mb-1">Rumus IKU 2:</p>
   448→                              <p>Persentase = Σ(n × k) / t × 100%</p>
   449→                              <p>n: Jumlah responden per kategori; k: Konstanta bobot; t: Total responden</p>
   450→                            </div>
   451→                          </div>
   452→                        </div>
   453→                      )}
   454→                      {iku.id === "iku3" && (
   455→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   456→                          <div className="flex items-start gap-2">
