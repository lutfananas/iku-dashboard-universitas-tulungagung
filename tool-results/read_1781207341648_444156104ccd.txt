    87→// ============ MAIN COMPONENT ============
    88→export default function IKUDashboard() {
    89→  const [navSelection, setNavSelection] = useState<NavSelection>({ mode: "global", id: "global" });
    90→  const [tahun, setTahun] = useState(2025);
    91→  const [allData, setAllData] = useState<IkuRecord[]>([]);
    92→  const [loading, setLoading] = useState(true);
    93→  const [expandedFakultas, setExpandedFakultas] = useState<Set<string>>(new Set());
    94→  const [saving, setSaving] = useState<string | null>(null);
    95→  const [saveSuccess, setSaveSuccess] = useState<string | null>(null);
    96→  const [mainTab, setMainTab] = useState<string>("dashboard");
    97→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
    98→
    99→  // Form state for current prodi+tahun
   100→  const [formData, setFormData] = useState<Record<string, Record<string, number | string>>>({});
   101→
   102→  // localStorage key
   103→  const STORAGE_KEY = "iku-unita-data";
   104→
   105→  // Load data from localStorage
   106→  const loadDataFromStorage = useCallback((): IkuRecord[] => {
   107→    try {
   108→      const stored = localStorage.getItem(STORAGE_KEY);
   109→      if (stored) {
   110→        const parsed = JSON.parse(stored);
   111→        if (Array.isArray(parsed)) return parsed;
   112→      }
   113→    } catch (e) {
   114→      console.error("localStorage read error:", e);
   115→    }
   116→    return [];
   117→  }, []);
   118→
   119→  // Save data to localStorage
   120→  const saveDataToStorage = useCallback((data: IkuRecord[]) => {
   121→    try {
   122→      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
   123→    } catch (e) {
   124→      console.error("localStorage write error:", e);
   125→    }
   126→  }, []);
   127→
   128→  // Fetch data (localStorage primary, API fallback)
   129→  const fetchData = useCallback(async () => {
   130→    try {
   131→      // Try API first
   132→      const res = await fetch(`/api/iku?tahun=${tahun}`);
   133→      if (res.ok) {
   134→        const data = await res.json();
   135→        if (Array.isArray(data) && data.length > 0) {
   136→          setAllData(data);
   137→          saveDataToStorage(data);
   138→          setLoading(false);
   139→          return;
   140→        }
   141→      }
   142→    } catch (e) {
   143→      console.error("API fetch error:", e);
   144→    }
   145→
   146→    // Fallback to localStorage
   147→    const stored = loadDataFromStorage();
   148→    const filtered = stored.filter((d: IkuRecord) => d.tahun === tahun);
   149→    if (filtered.length > 0) {
   150→      setAllData(filtered);
   151→    } else {
   152→      // Use all stored data (cross-year)
   153→      setAllData(stored);
   154→    }
   155→    setLoading(false);
   156→  }, [tahun, saveDataToStorage, loadDataFromStorage]);
   157→
   158→  useEffect(() => { fetchData(); }, [fetchData]);
   159→
   160→  // Get data for a specific prodi
   161→  const getProdiData = useCallback((prodiId: string): IkuRecord | undefined => {
   162→    return allData.find((d) => d.prodi === prodiId);
   163→  }, [allData]);
   164→
   165→  // Initialize form data when prodi/tahun changes
   166→  useEffect(() => {
   167→    if (navSelection.mode !== "prodi") return;
   168→    const prodiData = getProdiData(navSelection.id);
   169→    const newFormData: Record<string, Record<string, number | string>> = {};
   170→
   171→    IKU_LIST.forEach((iku) => {
   172→      const existingData = prodiData?.[iku.id as keyof IkuRecord] as Record<string, unknown> | null;
   173→      const fields = IKU_FIELDS[iku.id];
   174→      const ikuFormData: Record<string, number | string> = {};
   175→      fields.forEach((field) => {
   176→        if (existingData && existingData[field.key] !== undefined && existingData[field.key] !== null) {
   177→          ikuFormData[field.key] = existingData[field.key] as number | string;
   178→        } else {
   179→          ikuFormData[field.key] = field.defaultValue ?? (field.type === "number" ? 0 : "");
   180→        }
   181→      });
   182→      newFormData[iku.id] = ikuFormData;
   183→    });
   184→
   185→    setFormData(newFormData);
   186→  }, [navSelection, tahun, allData, getProdiData]);
   187→
   188→  // Save handler (localStorage primary, API sync)
   189→  const handleSave = async (ikuId: string) => {
   190→    if (navSelection.mode !== "prodi") return;
   191→    const prodi = getProdiById(navSelection.id);
   192→    if (!prodi) return;
   193→
   194→    setSaving(ikuId);
   195→    try {
   196→      // Update local state immediately
   197→      const existingIdx = allData.findIndex((d) => d.prodi === prodi.id && d.tahun === tahun);
   198→      const ikuData = formData[ikuId];
   199→
   200→      let updatedData: IkuRecord[];
   201→      if (existingIdx >= 0) {
   202→        updatedData = [...allData];
   203→        updatedData[existingIdx] = { ...updatedData[existingIdx], [ikuId]: ikuData };
   204→      } else {
   205→        const newRecord: IkuRecord = {
   206→          id: `local-${Date.now()}`,
   207→          prodi: prodi.id,
   208→          fakultas: prodi.fakultasId,
   209→          tahun,
   210→          iku1: null, iku2: null, iku3: null, iku5: null, iku7: null, iku9: null, iku12: null,
   211→          [ikuId]: ikuData,
   212→        };
   213→        updatedData = [...allData, newRecord];
   214→      }
   215→
   216→      setAllData(updatedData);
   217→      saveDataToStorage(updatedData);
   218→      setSaveSuccess(ikuId);
   219→      setTimeout(() => setSaveSuccess(null), 2000);
   220→      toast.success("Data berhasil disimpan", { description: `${IKU_LIST.find((i) => i.id === ikuId)?.label} - ${prodi.nama} (${tahun})` });
   221→
   222→      // Try API sync in background (non-blocking)
   223→      fetch("/api/iku", {
   224→        method: "POST",
   225→        headers: { "Content-Type": "application/json" },
   226→        body: JSON.stringify({
   227→          prodi: prodi.id,
   228→          fakultas: prodi.fakultasId,
   229→          tahun,
   230→          ikuId,
   231→          data: ikuData,
   232→        }),
   233→      }).catch(() => { /* API sync failed silently - data is safe in localStorage */ });
   234→    } catch (e) {
   235→      console.error("Save error:", e);
   236→    } finally {
   237→      setSaving(null);
   238→    }
   239→  };
   240→
   241→  // Form input change
   242→  const handleFormChange = (ikuId: string, key: string, value: number | string) => {
   243→    setFormData((prev) => ({
   244→      ...prev,
   245→      [ikuId]: { ...prev[ikuId], [key]: value },
   246→    }));
   247→  };
   248→
   249→  // Toggle fakultas expansion
   250→  const toggleFakultas = (fakultasId: string) => {
   251→    setExpandedFakultas((prev) => {
   252→      const next = new Set(prev);
   253→      if (next.has(fakultasId)) next.delete(fakultasId);
   254→      else next.add(fakultasId);
   255→      return next;
   256→    });
   257→  };
   258→
   259→  // ============ CALCULATION HELPERS ============
   260→  const calcProdiValue = (ikuId: string, prodiId: string): number => {
   261→    const prodiData = getProdiData(prodiId);
   262→    if (!prodiData) return 0;
   263→    const data = prodiData[ikuId as keyof IkuRecord];
   264→    if (!data || !hasData(data as Record<string, unknown>)) return 0;
   265→
   266→    const prodi = getProdiById(prodiId);
   267→    switch (ikuId) {
   268→      case "iku1": return calcAEE_Realisation(data as Iku1Data);
   269→      case "iku2": return calcIKU2(data as Iku2Data);
   270→      case "iku3": return calcIKU3(data as Iku3Data);
   271→      case "iku5": return calcIKU5(data as Iku5Data);
   272→      case "iku7": return calcIKU7(data as Iku7Data);
   273→      case "iku9": return calcIKU9(data as Iku9Data);
   274→      case "iku12": return calcIKU12_Skor(data as Iku12Data);
   275→      default: return 0;
   276→    }
   277→  };
   278→
   279→  const calcAggregatedValue = (ikuId: string, prodiIds: string[]): number => {
   280→    if (prodiIds.length === 0) return 0;
   281→
   282→    // Hitung skor individu setiap prodi (yang belum isi = 0)
   283→    const scores = prodiIds.map((pid) => calcProdiValue(ikuId, pid));
   284→
   285→    // Rata-rata dari SEMUA prodi (prodi tanpa data otomatis 0)
   286→    return scores.reduce((a, b) => a + b, 0) / prodiIds.length;
   287→  };
   288→
   289→  // Hitung berapa prodi yang sudah mengisi data untuk IKU tertentu
   290→  const getIkuCoverage = (ikuId: string, prodiIds: string[]): { filled: number; total: number } => {
   291→    const filled = prodiIds.filter((pid) => calcProdiValue(ikuId, pid) > 0).length;
   292→    return { filled, total: prodiIds.length };
   293→  };
   294→
   295→  const getProdiHasData = (prodiId: string): boolean => {
   296→    const d = getProdiData(prodiId);
   297→    if (!d) return false;
   298→    return IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>));
   299→  };
   300→
   301→  const getFakultasHasData = (fakultasId: string): boolean => {
   302→    return getProdiByFakultas(fakultasId).some((p) => getProdiHasData(p.id));
   303→  };
   304→
   305→  // ============ RENDER: SIDEBAR ============
   306→  const renderSidebar = () => (
   307→    <aside className="w-64 lg:w-72 shrink-0 sidebar-pattern overflow-y-auto custom-scrollbar" style={{ background: "linear-gradient(180deg, #0F172A 0%, #1E293B 100%)" }}>
   308→      {/* University Header with Logo */}
   309→      <div className="p-5 text-center border-b border-white/10">
   310→        <div className="flex justify-center mb-3">
   311→          <div className="w-[60px] h-[60px] rounded-xl bg-white/10 p-1.5 flex items-center justify-center backdrop-blur-sm border border-white/10">
   312→            <img src="/logo-unita.png" alt="UNITA" className="w-full h-full object-contain" />
   313→          </div>
   314→        </div>
   315→        <h2 className="font-bold text-sm text-white tracking-wide">UNIVERSITAS TULUNGAGUNG</h2>
   316→        <p className="text-[11px] mt-1 font-semibold" style={{ color: GOLD }}>Dashboard IKU</p>
   317→      </div>
   318→
   319→      {/* Year Selector */}
   320→      <div className="p-3 border-b border-white/10">
   321→        <Label className="text-[10px] font-medium text-white/40 mb-1.5 block uppercase tracking-widest">Tahun Akademik</Label>
   322→        <Select value={tahun.toString()} onValueChange={(v) => setTahun(parseInt(v))}>
   323→          <SelectTrigger className="h-9 text-sm border-white/10 bg-white/5 text-white hover:bg-white/10 transition-colors" style={{ borderColor: "rgba(255,255,255,0.1)" }}>
   324→            <Calendar className="w-3.5 h-3.5 mr-1.5 text-white/50" />
   325→            <SelectValue />
   326→          </SelectTrigger>
   327→          <SelectContent>
   328→            {[2025, 2026, 2027, 2028, 2029, 2030].map((y) => (
   329→              <SelectItem key={y} value={y.toString()}>{y}</SelectItem>
   330→            ))}
   331→          </SelectContent>
   332→        </Select>
   333→      </div>
   334→
   335→      {/* Navigation Tree */}
   336→      <nav className="p-2">
   337→        {/* Global */}
   338→        <button
   339→          onClick={() => { setNavSelection({ mode: "global", id: "global" }); setMobileMenuOpen(false); }}
   340→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-xl text-sm font-medium transition-all duration-200 ${
   341→            navSelection.mode === "global"
   342→              ? "text-white shadow-lg"
   343→              : "text-white/60 hover:text-white hover:bg-white/5"
   344→          }`}
   345→          style={navSelection.mode === "global" ? { background: "linear-gradient(135deg, #B91C1C, #8B1A1A)", boxShadow: "0 4px 15px rgba(185,28,28,0.3)" } : {}}
   346→        >
   347→          <BarChart3 className="w-4 h-4 shrink-0" />
   348→          <span>Global Universitas</span>
   349→          {allData.length > 0 && (
   350→            <Badge className="ml-auto text-[10px] px-1.5 py-0" style={{ backgroundColor: "rgba(5,150,105,0.8)", color: "white" }}>
   351→              {allData.filter((d) => IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>))).length} prodi
   352→            </Badge>
   353→          )}
   354→        </button>
   355→
   356→        <Separator className="my-2 bg-white/10" />
   357→
   358→        {/* Fakultas List */}
   359→        {FAKULTAS_LIST.map((fakultas) => {
   360→          const isExpanded = expandedFakultas.has(fakultas.id);
   361→          const isSelected = navSelection.mode === "fakultas" && navSelection.id === fakultas.id;
   362→          const hasD = getFakultasHasData(fakultas.id);
   363→
   364→          return (
   365→            <div key={fakultas.id} className="mb-0.5">
   366→              <button
   367→                onClick={() => {
   368→                  toggleFakultas(fakultas.id);
   369→                  setNavSelection({ mode: "fakultas", id: fakultas.id });
   370→                }}
   371→                className={`w-full flex items-center gap-2 px-3 py-2 rounded-xl text-sm transition-all duration-200 ${
   372→                  isSelected ? "text-white bg-white/10 font-semibold" : "text-white/60 hover:text-white hover:bg-white/5"
   373→                }`}
   374→              >
   375→                {isExpanded ? <ChevronDown className="w-3.5 h-3.5 shrink-0" /> : <ChevronRight className="w-3.5 h-3.5 shrink-0" />}
   376→                <span className="truncate flex-1 text-left">{fakultas.nama.replace("Fakultas ", "F. ")}</span>
   377→                {hasD && <div className="w-2 h-2 rounded-full shrink-0" style={{ backgroundColor: TEAL }} />}
   378→              </button>
   379→
   380→              {isExpanded && (
   381→                <div className="ml-4 mt-0.5 space-y-0.5 pl-2 border-l border-white/10">
   382→                  {fakultas.prodiList.map((prodi) => {
   383→                    const isProdiSelected = navSelection.mode === "prodi" && navSelection.id === prodi.id;
   384→                    const prodiHasD = getProdiHasData(prodi.id);
   385→                    return (
   386→                      <button
   387→                        key={prodi.id}
   388→                        onClick={() => { setNavSelection({ mode: "prodi", id: prodi.id }); setMobileMenuOpen(false); }}
   389→                        className={`w-full flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs transition-all duration-200 ${
   390→                          isProdiSelected
   391→                            ? "text-white font-medium"
   392→                            : "text-white/40 hover:text-white hover:bg-white/5"
   393→                        }`}
   394→                        style={isProdiSelected ? { background: "linear-gradient(135deg, #1E40AF, #1E3A5F)", boxShadow: "0 2px 10px rgba(30,64,175,0.3)" } : {}}
   395→                      >
   396→                        <GraduationCap className="w-3.5 h-3.5 shrink-0" />
   397→                        <span className="truncate flex-1 text-left">{prodi.nama}</span>
   398→                        {prodiHasD && <div className="w-1.5 h-1.5 rounded-full shrink-0" style={{ backgroundColor: TEAL }} />}
   399→                      </button>
   400→                    );
   401→                  })}
   402→                </div>
   403→              )}
   404→            </div>
   405→          );
   406→        })}
   407→      </nav>
   408→    </aside>
   409→  );
   410→
   411→  // ============ RENDER: IKU FORM FOR PRODI ============
   412→  const renderProdiForm = () => {
   413→    const prodi = getProdiById(navSelection.id);
   414→    if (!prodi) return null;
   415→    const fakultas = getFakultasById(prodi.fakultasId);
   416→
   417→    return (
   418→      <div className="space-y-6">
   419→        {/* Header */}
   420→        <Card className="glass-card overflow-hidden transition-all duration-300 hover:shadow-xl" style={{ borderTop: `3px solid ${PRIMARY}` }}>
   421→          <CardContent className="p-5">
   422→            <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">
   423→              <div>
   424→                <h2 className="text-lg font-bold gradient-text">{prodi.nama}</h2>
   425→                <p className="text-sm text-slate-500">{fakultas?.nama} • Tahun Akademik {tahun}</p>
   426→              </div>
   427→              <Badge className="w-fit text-xs border-0 font-bold" style={{ background: "linear-gradient(135deg, #1E40AF, #1E3A5F)", color: "white" }}>{prodi.jenjang}</Badge>
   428→            </div>
   429→          </CardContent>
   430→        </Card>
   431→
   432→        {/* IKU Tabs */}
   433→        <Tabs defaultValue="iku1" className="space-y-4">
   434→          <TabsList className="iku-tabs w-full flex h-auto p-1.5 rounded-xl gap-1 overflow-x-auto" style={{ background: "linear-gradient(135deg, rgba(15,23,42,0.05), rgba(30,41,59,0.08))" }}>
   435→            {IKU_LIST.map((iku) => (
   436→              <TabsTrigger key={iku.id} value={iku.id} className="text-xs font-semibold px-3 py-2 rounded-lg whitespace-nowrap data-[state=active]:shadow-lg transition-all duration-200" style={{ color: IKU_COLORS[iku.id] }}>
   437→                {iku.label}
   438→              </TabsTrigger>
   439→            ))}
   440→          </TabsList>
   441→
   442→          {IKU_LIST.map((iku) => {
   443→            const fields = IKU_FIELDS[iku.id];
   444→            const currentData = formData[iku.id] || {};
   445→            const isSaving = saving === iku.id;
   446→            const isSuccess = saveSuccess === iku.id;
   447→
   448→            // Calculate preview
   449→            let previewValue = 0;
   450→            let previewLabel = "";
   451→            try {
