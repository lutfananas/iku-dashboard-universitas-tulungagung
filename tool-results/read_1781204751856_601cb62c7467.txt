    71→// ============ MAIN COMPONENT ============
    72→export default function IKUDashboard() {
    73→  const [navSelection, setNavSelection] = useState<NavSelection>({ mode: "global", id: "global" });
    74→  const [tahun, setTahun] = useState(2025);
    75→  const [allData, setAllData] = useState<IkuRecord[]>([]);
    76→  const [loading, setLoading] = useState(true);
    77→  const [expandedFakultas, setExpandedFakultas] = useState<Set<string>>(new Set());
    78→  const [saving, setSaving] = useState<string | null>(null);
    79→  const [saveSuccess, setSaveSuccess] = useState<string | null>(null);
    80→  const [mainTab, setMainTab] = useState<string>("dashboard");
    81→
    82→  // Form state for current prodi+tahun
    83→  const [formData, setFormData] = useState<Record<string, Record<string, number | string>>>({});
    84→
    85→  // Fetch data
    86→  const fetchData = useCallback(async () => {
    87→    try {
    88→      const res = await fetch(`/api/iku?tahun=${tahun}`);
    89→      const data = await res.json();
    90→      setAllData(data);
    91→    } catch (e) {
    92→      console.error("Fetch error:", e);
    93→    } finally {
    94→      setLoading(false);
    95→    }
    96→  }, [tahun]);
    97→
    98→  useEffect(() => { fetchData(); }, [fetchData]);
    99→
   100→  // Get data for a specific prodi
   101→  const getProdiData = useCallback((prodiId: string): IkuRecord | undefined => {
   102→    return allData.find((d) => d.prodi === prodiId);
   103→  }, [allData]);
   104→
   105→  // Initialize form data when prodi/tahun changes
   106→  useEffect(() => {
   107→    if (navSelection.mode !== "prodi") return;
   108→    const prodiData = getProdiData(navSelection.id);
   109→    const newFormData: Record<string, Record<string, number | string>> = {};
   110→
   111→    IKU_LIST.forEach((iku) => {
   112→      const existingData = prodiData?.[iku.id as keyof IkuRecord] as Record<string, unknown> | null;
   113→      const fields = IKU_FIELDS[iku.id];
   114→      const ikuFormData: Record<string, number | string> = {};
   115→      fields.forEach((field) => {
   116→        if (existingData && existingData[field.key] !== undefined && existingData[field.key] !== null) {
   117→          ikuFormData[field.key] = existingData[field.key] as number | string;
   118→        } else {
   119→          ikuFormData[field.key] = field.defaultValue ?? (field.type === "number" ? 0 : "");
   120→        }
   121→      });
   122→      newFormData[iku.id] = ikuFormData;
   123→    });
   124→
   125→    setFormData(newFormData);
   126→  }, [navSelection, tahun, allData, getProdiData]);
   127→
   128→  // Save handler
   129→  const handleSave = async (ikuId: string) => {
   130→    if (navSelection.mode !== "prodi") return;
   131→    const prodi = getProdiById(navSelection.id);
   132→    if (!prodi) return;
   133→
   134→    setSaving(ikuId);
   135→    try {
   136→      const res = await fetch("/api/iku", {
   137→        method: "POST",
   138→        headers: { "Content-Type": "application/json" },
   139→        body: JSON.stringify({
   140→          prodi: prodi.id,
   141→          fakultas: prodi.fakultasId,
   142→          tahun,
   143→          ikuId,
   144→          data: formData[ikuId],
   145→        }),
   146→      });
   147→      if (res.ok) {
   148→        setSaveSuccess(ikuId);
   149→        setTimeout(() => setSaveSuccess(null), 2000);
   150→        await fetchData();
   151→        toast.success("Data berhasil disimpan", { description: `${IKU_LIST.find((i) => i.id === ikuId)?.label} - ${prodi.nama} (${tahun})` });
   152→      }
   153→    } catch (e) {
   154→      console.error("Save error:", e);
   155→    } finally {
   156→      setSaving(null);
   157→    }
   158→  };
   159→
   160→  // Form input change
   161→  const handleFormChange = (ikuId: string, key: string, value: number | string) => {
   162→    setFormData((prev) => ({
   163→      ...prev,
   164→      [ikuId]: { ...prev[ikuId], [key]: value },
   165→    }));
   166→  };
   167→
   168→  // Toggle fakultas expansion
   169→  const toggleFakultas = (fakultasId: string) => {
   170→    setExpandedFakultas((prev) => {
   171→      const next = new Set(prev);
   172→      if (next.has(fakultasId)) next.delete(fakultasId);
   173→      else next.add(fakultasId);
   174→      return next;
   175→    });
   176→  };
   177→
   178→  // ============ CALCULATION HELPERS ============
   179→  const calcProdiValue = (ikuId: string, prodiId: string): number => {
   180→    const prodiData = getProdiData(prodiId);
   181→    if (!prodiData) return 0;
   182→    const data = prodiData[ikuId as keyof IkuRecord];
   183→    if (!data || !hasData(data as Record<string, unknown>)) return 0;
   184→
   185→    const prodi = getProdiById(prodiId);
   186→    switch (ikuId) {
   187→      case "iku1": return calcAEE_Realisation(data as Iku1Data);
   188→      case "iku2": return calcIKU2(data as Iku2Data);
   189→      case "iku3": return calcIKU3(data as Iku3Data);
   190→      case "iku5": return calcIKU5(data as Iku5Data);
   191→      case "iku7": return calcIKU7(data as Iku7Data);
   192→      case "iku9": return calcIKU9(data as Iku9Data);
   193→      case "iku12": return calcIKU12_Skor(data as Iku12Data);
   194→      default: return 0;
   195→    }
   196→  };
   197→
   198→  const calcAggregatedValue = (ikuId: string, prodiIds: string[]): number => {
   199→    if (prodiIds.length === 0) return 0;
   200→
   201→    // Hitung skor individu setiap prodi (yang belum isi = 0)
   202→    const scores = prodiIds.map((pid) => calcProdiValue(ikuId, pid));
   203→
   204→    // Rata-rata dari SEMUA prodi (prodi tanpa data otomatis 0)
   205→    return scores.reduce((a, b) => a + b, 0) / prodiIds.length;
   206→  };
   207→
   208→  // Hitung berapa prodi yang sudah mengisi data untuk IKU tertentu
   209→  const getIkuCoverage = (ikuId: string, prodiIds: string[]): { filled: number; total: number } => {
   210→    const filled = prodiIds.filter((pid) => calcProdiValue(ikuId, pid) > 0).length;
   211→    return { filled, total: prodiIds.length };
   212→  };
   213→
   214→  const getProdiHasData = (prodiId: string): boolean => {
   215→    const d = getProdiData(prodiId);
   216→    if (!d) return false;
   217→    return IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>));
   218→  };
   219→
   220→  const getFakultasHasData = (fakultasId: string): boolean => {
   221→    return getProdiByFakultas(fakultasId).some((p) => getProdiHasData(p.id));
   222→  };

   232→        </div>
   233→        <p className="text-xs text-white/60">Dashboard IKU Perguruan Tinggi</p>
   234→      </div>
   235→
   236→      {/* Year Selector */}

   238→        <Label className="text-xs font-medium text-slate-500 mb-1.5 block">Tahun Akademik</Label>
   239→        <Select value={tahun.toString()} onValueChange={(v) => setTahun(parseInt(v))}>
   240→          <SelectTrigger className="h-9 text-sm border-slate-200">
   241→            <Calendar className="w-3.5 h-3.5 mr-1.5 text-slate-400" />
   242→            <SelectValue />
   243→          </SelectTrigger>
   244→          <SelectContent>
   245→            {[2025, 2026, 2027, 2028, 2029, 2030].map((y) => (
   246→              <SelectItem key={y} value={y.toString()}>{y}</SelectItem>
   247→            ))}
   248→          </SelectContent>
   249→        </Select>
   250→      </div>
   251→
   252→      {/* Navigation Tree */}
   253→      <nav className="p-2">
   254→        {/* Global */}
   255→        <button
   256→          onClick={() => setNavSelection({ mode: "global", id: "global" })}
   257→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-all ${
   258→            navSelection.mode === "global"
   259→              ? "bg-navy text-white shadow-md"
   260→              : "text-slate-600 hover:bg-slate-50"
   261→          }`}
   262→        >
   263→          <BarChart3 className="w-4 h-4 shrink-0" />
   264→          <span>Global Universitas</span>
   265→          {allData.length > 0 && (
   266→            <Badge className="ml-auto text-[10px] bg-teal text-white px-1.5 py-0">
   267→              {allData.filter((d) => IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>))).length} prodi
   268→            </Badge>
   269→          )}
   270→        </button>
   271→
   272→        <Separator className="my-2" />
   273→
   274→        {/* Fakultas List */}
   275→        {FAKULTAS_LIST.map((fakultas) => {
   276→          const isExpanded = expandedFakultas.has(fakultas.id);
   277→          const isSelected = navSelection.mode === "fakultas" && navSelection.id === fakultas.id;
   278→          const hasD = getFakultasHasData(fakultas.id);
   279→
   280→          return (
   281→            <div key={fakultas.id} className="mb-0.5">
   282→              <button
   283→                onClick={() => {
   284→                  toggleFakultas(fakultas.id);
   285→                  setNavSelection({ mode: "fakultas", id: fakultas.id });
   286→                }}
   287→                className={`w-full flex items-center gap-2 px-3 py-2 rounded-lg text-sm transition-all ${
   288→                  isSelected ? "bg-navy/10 text-navy font-semibold" : "text-slate-600 hover:bg-slate-50"
   289→                }`}
   290→              >
   291→                {isExpanded ? <ChevronDown className="w-3.5 h-3.5 shrink-0" /> : <ChevronRight className="w-3.5 h-3.5 shrink-0" />}
   292→                <span className="truncate flex-1 text-left">{fakultas.nama.replace("Fakultas ", "F. ")}</span>
   293→                {hasD && <div className="w-2 h-2 rounded-full bg-teal shrink-0" />}
   294→              </button>
   295→
   296→              {isExpanded && (
   297→                <div className="ml-5 mt-0.5 space-y-0.5">
   298→                  {fakultas.prodiList.map((prodi) => {
   299→                    const isProdiSelected = navSelection.mode === "prodi" && navSelection.id === prodi.id;
   300→                    const prodiHasD = getProdiHasData(prodi.id);
   301→                    return (
   302→                      <button
   303→                        key={prodi.id}
   304→                        onClick={() => setNavSelection({ mode: "prodi", id: prodi.id })}
   305→                        className={`w-full flex items-center gap-2 px-3 py-1.5 rounded-md text-xs transition-all ${
   306→                          isProdiSelected
   307→                            ? "bg-navy text-white font-medium"
   308→                            : "text-slate-500 hover:bg-slate-50 hover:text-slate-700"
   309→                        }`}
   310→                      >
   311→                        <GraduationCap className="w-3.5 h-3.5 shrink-0" />
   312→                        <span className="truncate flex-1 text-left">{prodi.nama}</span>
   313→                        {prodiHasD && <div className="w-1.5 h-1.5 rounded-full bg-teal shrink-0" />}
   314→                      </button>
   315→                    );
   316→                  })}
   317→                </div>
   318→              )}
   319→            </div>
   320→          );
   321→        })}
   322→      </nav>

   327→  const renderProdiForm = () => {

   329→    if (!prodi) return null;
   330→    const fakultas = getFakultasById(prodi.fakultasId);
   331→
   332→    return (
   333→      <div className="space-y-6">
   334→        {/* Header */}
   335→        <Card className="border-l-4 border-l-navy">
   336→          <CardContent className="p-5">
   337→            <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">
   338→              <div>
   339→                <h2 className="text-lg font-bold text-navy">{prodi.nama}</h2>
   340→                <p className="text-sm text-slate-500">{fakultas?.nama} • Tahun Akademik {tahun}</p>

   342→              <Badge variant="outline" className="w-fit text-xs">{prodi.jenjang}</Badge>
   343→            </div>
   344→          </CardContent>
   345→        </Card>
   346→
   347→        {/* IKU Tabs */}
   348→        <Tabs defaultValue="iku1" className="space-y-4">

   350→            {IKU_LIST.map((iku) => (
   351→              <TabsTrigger key={iku.id} value={iku.id} className="text-xs font-semibold px-3 py-2 rounded-md whitespace-nowrap data-[state=active]:shadow-md transition-all">
   352→                {iku.label}
   353→              </TabsTrigger>
   354→            ))}
   355→          </TabsList>
   356→
   357→          {IKU_LIST.map((iku) => {
   358→            const fields = IKU_FIELDS[iku.id];
   359→            const currentData = formData[iku.id] || {};
   360→            const isSaving = saving === iku.id;
   361→            const isSuccess = saveSuccess === iku.id;
   362→
   363→            // Calculate preview
   364→            let previewValue = 0;
   365→            let previewLabel = "";
   366→            try {
   367→              switch (iku.id) {
   368→                case "iku1": {
   369→                  const d = currentData as unknown as Iku1Data;
   370→                  previewValue = calcAEE_Realisation(d);
   371→                  previewLabel = `AEE Realisasi: ${previewValue.toFixed(2)}% | Tingkat Pencapaian: ${calcAEE_AchievementRate(d, prodi.jenjang).toFixed(2)}%`;
   372→                  break;
   373→                }
   374→                case "iku2": {
   375→                  previewValue = calcIKU2(currentData as unknown as Iku2Data);
   376→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   377→                  break;
   378→                }
   379→                case "iku3": {
   380→                  previewValue = calcIKU3(currentData as unknown as Iku3Data);
   381→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   382→                  break;
   383→                }
   384→                case "iku5": {
   385→                  previewValue = calcIKU5(currentData as unknown as Iku5Data);
   386→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   387→                  break;
   388→                }
   389→                case "iku7": {
   390→                  previewValue = calcIKU7(currentData as unknown as Iku7Data);
   391→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   392→                  break;
   393→                }
   394→                case "iku9": {
   395→                  previewValue = calcIKU9(currentData as unknown as Iku9Data);
   396→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   397→                  break;
   398→                }
   399→                case "iku12": {
   400→                  previewValue = calcIKU12_Skor(currentData as unknown as Iku12Data);
   401→                  previewLabel = `Skor: ${previewValue.toFixed(2)}`;
   402→                  break;
   403→                }
   404→              }
   405→            } catch { /* empty */ }
   406→
   407→            return (
   408→              <TabsContent key={iku.id} value={iku.id} className="mt-4">
   409→                <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   410→                  {/* Input Form */}
   411→                  <Card className="lg:col-span-2 border border-slate-200">
   412→                    <CardHeader className="pb-3">
   413→                      <CardTitle className="text-base font-semibold text-navy">{iku.title}</CardTitle>
   414→                      <CardDescription className="text-xs">Masukkan data untuk {prodi.nama} tahun {tahun}</CardDescription>
   415→                    </CardHeader>
   416→                    <CardContent className="space-y-4">
   417→                      {iku.id === "iku1" && (
   418→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   419→                          <div className="flex items-start gap-2">
   420→                            <Info className="w-4 h-4 text-navy mt-0.5 shrink-0" />
   421→                            <div className="text-xs text-slate-600">
   422→                              <p className="font-semibold text-navy mb-1">Rumus IKU 1 (AEE PT):</p>
   423→                              <p>AEE Realisasi = (Lulusan Tepat Waktu / Total Mahasiswa) × 100%</p>
   424→                              <p>Tingkat Pencapaian = (AEE Realisasi / AEE Ideal) × 100%</p>
   425→                              <p>AEE Ideal {prodi.jenjang} = {(() => { const v = { D3: 33, D4: 25, S1: 25, S2: 50, S3: 33 } as Record<string, number>; return v[prodi.jenjang] || 25; })()}%</p>
   426→                            </div>
   427→                          </div>
   428→                        </div>
   429→                      )}
   430→                      {iku.id === "iku2" && (
   431→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   432→                          <div className="flex items-start gap-2">
   433→                            <Info className="w-4 h-4 text-navy mt-0.5 shrink-0" />
   434→                            <div className="text-xs text-slate-600">
   435→                              <p className="font-semibold text-navy mb-1">Rumus IKU 2:</p>

   439→                            </div>
   440→                          </div>
   441→                        </div>
   442→                      )}
   443→                      {iku.id === "iku3" && (
   444→                        <div className="bg-slate-50 rounded-lg p-3 mb-3 border border-slate-100">
   445→                          <div className="flex items-start gap-2">
   446→                            <Info className="w-4 h-4 text-navy mt-0.5 shrink-0" />
   447→                            <div className="text-xs text-slate-600">
