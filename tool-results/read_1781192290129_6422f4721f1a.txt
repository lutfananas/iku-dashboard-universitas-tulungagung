    85→    85→    } catch (e) {

    87→    87→    } finally {
    88→    88→      setLoading(false);
    89→    89→    }
    90→    90→  }, [tahun]);
    91→    91→
    92→    92→  useEffect(() => { fetchData(); }, [fetchData]);

    97→    97→  }, [allData]);

    99→    99→  // Initialize form data when prodi/tahun changes
   100→   100→  useEffect(() => {
   101→   101→    if (navSelection.mode !== "prodi") return;
   102→   102→    const prodiData = getProdiData(navSelection.id);
   103→   103→    const newFormData: Record<string, Record<string, number | string>> = {};
   104→   104→
   105→   105→    IKU_LIST.forEach((iku) => {
   106→   106→      const existingData = prodiData?.[iku.id as keyof IkuRecord] as Record<string, unknown> | null;
   107→   107→      const fields = IKU_FIELDS[iku.id];
   108→   108→      const ikuFormData: Record<string, number | string> = {};
   109→   109→      fields.forEach((field) => {
   110→   110→        if (existingData && existingData[field.key] !== undefined && existingData[field.key] !== null) {
   111→   111→          ikuFormData[field.key] = existingData[field.key] as number | string;
   112→   112→        } else {
   113→   113→          ikuFormData[field.key] = field.defaultValue ?? (field.type === "number" ? 0 : "");
   114→   114→        }
   115→   115→      });
   116→   116→      newFormData[iku.id] = ikuFormData;
   117→   117→    });
   118→   118→
   119→   119→    setFormData(newFormData);
   120→   120→  }, [navSelection, tahun, allData, getProdiData]);
   121→   121→
   122→   122→  // Save handler
   123→   123→  const handleSave = async (ikuId: string) => {
   124→   124→    if (navSelection.mode !== "prodi") return;
   125→   125→    const prodi = getProdiById(navSelection.id);
   126→   126→    if (!prodi) return;
   127→   127→
   128→   128→    setSaving(ikuId);
   129→   129→    try {
   130→   130→      const res = await fetch("/api/iku", {
   131→   131→        method: "POST",
   132→   132→        headers: { "Content-Type": "application/json" },
   133→   133→        body: JSON.stringify({
   134→   134→          prodi: prodi.id,
   135→   135→          fakultas: prodi.fakultasId,
   136→   136→          tahun,
   137→   137→          ikuId,
   138→   138→          data: formData[ikuId],
   139→   139→        }),
   140→   140→      });

   142→   142→        setSaveSuccess(ikuId);
   143→   143→        setTimeout(() => setSaveSuccess(null), 2000);
   144→   144→        await fetchData();
   145→   145→        toast.success("Data berhasil disimpan", { description: `${IKU_LIST.find((i) => i.id === ikuId)?.label} - ${prodi.nama} (${tahun})` });
   146→   146→      }
   147→   147→    } catch (e) {
   148→   148→      console.error("Save error:", e);
   149→   149→    } finally {
   150→   150→      setSaving(null);
   151→   151→    }

   153→   153→
   154→   154→  // Form input change

   157→   157→      ...prev,
   158→   158→      [ikuId]: { ...prev[ikuId], [key]: value },
   159→   159→    }));

   166→   166→      if (next.has(fakultasId)) next.delete(fakultasId);

   169→   169→    });

   175→   175→    if (!prodiData) return 0;
   176→   176→    const data = prodiData[ikuId as keyof IkuRecord];

   178→   178→
   179→   179→    const prodi = getProdiById(prodiId);
   180→   180→    switch (ikuId) {
   181→   181→      case "iku1": return calcAEE_AchievementRate(data as Iku1Data, prodi?.jenjang || "S1");
   182→   182→      case "iku2": return calcIKU2(data as Iku2Data);
   183→   183→      case "iku3": return calcIKU3(data as Iku3Data);
   184→   184→      case "iku5": return calcIKU5(data as Iku5Data);
   185→   185→      case "iku7": return calcIKU7(data as Iku7Data);
   186→   186→      case "iku9": return calcIKU9(data as Iku9Data);
   187→   187→      case "iku12": return calcIKU12_Skor(data as Iku12Data);
   188→   188→      default: return 0;
   189→   189→    }
   190→   190→  };
   191→   191→
   192→   192→  const calcAggregatedValue = (ikuId: string, prodiIds: string[]): number => {

   200→   200→        if (data && hasData(data as Record<string, unknown>)) {
   201→   201→          dataList.push(data as Record<string, unknown>);
   202→   202→          const prodi = getProdiById(pid);
   203→   203→          if (prodi) prodiJenjangs.push(prodi.jenjang);
   204→   204→        }
   205→   205→      }
   206→   206→    });
   207→   207→
   208→   208→    if (dataList.length === 0) return 0;
   209→   209→
   210→   210→    switch (ikuId) {
   211→   211→      case "iku1": {
   212→   212→        const validData = dataList.map((d, i) => ({ data: d as Iku1Data, jenjang: prodiJenjangs[i] || "S1" }));
   213→   213→        return calcAEE_PT(validData);
   214→   214→      }
   215→   215→      case "iku2": return calcIKU2_Aggregate(dataList as Iku2Data[]);
   216→   216→      case "iku3": return calcIKU3_Aggregate(dataList as Iku3Data[]);
   217→   217→      case "iku5": return calcIKU5_Aggregate(dataList as Iku5Data[]);
   218→   218→      case "iku7": return calcIKU7_Aggregate(dataList as Iku7Data[]);
   219→   219→      case "iku9": return calcIKU9_Aggregate(dataList as Iku9Data[]);
   220→   220→      case "iku12": return calcIKU12_Aggregate(dataList as Iku12Data[]);
   221→   221→      default: return 0;
   222→   222→    }
   223→   223→  };
   224→   224→
   225→   225→  const getProdiHasData = (prodiId: string): boolean => {
   226→   226→    const d = getProdiData(prodiId);
   227→   227→    if (!d) return false;
   228→   228→    return IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>));
   229→   229→  };
   230→   230→
   231→   231→  const getFakultasHasData = (fakultasId: string): boolean => {

   233→   233→  };
   234→   234→

   240→   240→        <div className="flex items-center gap-2 mb-1">

   253→   253→            <SelectValue />
   254→   254→          </SelectTrigger>

   256→   256→            {[2025, 2026, 2027, 2028, 2029, 2030].map((y) => (

   259→   259→          </SelectContent>

   262→   262→
   263→   263→      {/* Navigation Tree */}
   264→   264→      <nav className="p-2">
   265→   265→        {/* Global */}
   266→   266→        <button
   267→   267→          onClick={() => setNavSelection({ mode: "global", id: "global" })}
   268→   268→          className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm font-medium transition-all ${
   269→   269→            navSelection.mode === "global"
   270→   270→              ? "bg-navy text-white shadow-md"
   271→   271→              : "text-slate-600 hover:bg-slate-50"
   272→   272→          }`}
   273→   273→        >
   274→   274→          <BarChart3 className="w-4 h-4 shrink-0" />

   276→   276→          {allData.length > 0 && (
   277→   277→            <Badge className="ml-auto text-[10px] bg-teal text-white px-1.5 py-0">
   278→   278→              {allData.filter((d) => IKU_LIST.some((iku) => d[iku.id as keyof IkuRecord] !== null && hasData(d[iku.id as keyof IkuRecord] as Record<string, unknown>))).length} prodi
   279→   279→            </Badge>
   280→   280→          )}
   281→   281→        </button>

   289→   289→          const hasD = getFakultasHasData(fakultas.id);

   292→   292→            <div key={fakultas.id} className="mb-0.5">
   293→   293→              <button

   296→   296→                  setNavSelection({ mode: "fakultas", id: fakultas.id });
   297→   297→                }}
   298→   298→                className={`w-full flex items-center gap-2 px-3 py-2 rounded-lg text-sm transition-all ${
   299→   299→                  isSelected ? "bg-navy/10 text-navy font-semibold" : "text-slate-600 hover:bg-slate-50"

   325→   325→                      </button>

   328→   328→                </div>

   337→   337→  // ============ RENDER: IKU FORM FOR PRODI ============
   338→   338→  const renderProdiForm = () => {

   345→   345→        {/* Header */}
   346→   346→        <Card className="border-l-4 border-l-navy">

   348→   348→            <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">

   350→   350→                <h2 className="text-lg font-bold text-navy">{prodi.nama}</h2>

   354→   354→            </div>
   355→   355→          </CardContent>
   356→   356→        </Card>
   357→   357→
   358→   358→        {/* IKU Tabs */}
   359→   359→        <Tabs defaultValue="iku1" className="space-y-4">

   361→   361→            {IKU_LIST.map((iku) => (

   363→   363→                {iku.label}
   364→   364→              </TabsTrigger>
   365→   365→            ))}
   366→   366→          </TabsList>
   367→   367→
   368→   368→          {IKU_LIST.map((iku) => {
   369→   369→            const fields = IKU_FIELDS[iku.id];
   370→   370→            const currentData = formData[iku.id] || {};
   371→   371→            const isSaving = saving === iku.id;
   372→   372→            const isSuccess = saveSuccess === iku.id;
   373→   373→
   374→   374→            // Calculate preview
   375→   375→            let previewValue = 0;
   376→   376→            let previewLabel = "";
   377→   377→            try {
   378→   378→              switch (iku.id) {
   379→   379→                case "iku1": {
   380→   380→                  const d = currentData as unknown as Iku1Data;
   381→   381→                  previewValue = calcAEE_Realisation(d);

   387→   387→                  previewValue = calcIKU2(currentData as unknown as Iku2Data);
   388→   388→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   389→   389→                  break;
   390→   390→                }
   391→   391→                case "iku3": {
   392→   392→                  previewValue = calcIKU3(currentData as unknown as Iku3Data);
   393→   393→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   394→   394→                  break;
   395→   395→                }
   396→   396→                case "iku5": {
   397→   397→                  previewValue = calcIKU5(currentData as unknown as Iku5Data);
   398→   398→                  previewLabel = `Persentase: ${previewValue.toFixed(2)}%`;
   399→   399→                  break;
